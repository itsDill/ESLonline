# .htaccess Security Improvements

# ESL Fun Online - Enhanced Protection

# ============================================================================

# RECOMMENDED ADDITIONS TO CURRENT .htaccess

# ============================================================================

# Prevent access to sensitive file types (add to existing config)

<FilesMatch "\.(py|pyc|sh|bak|backup|tmp|swp|swo|~)$">
Order allow,deny
Deny from all # Also return 403 Forbidden instead of 404
</FilesMatch>

# Block access to .dev-files directory completely

<Directory ".dev-files">
Order allow,deny
Deny from all
</Directory>

# Block access to .venv directory

<Directory ".venv">
Order allow,deny
Deny from all
</Directory>

# ============================================================================

# ADDITIONAL SECURITY HEADERS (ENHANCED)

# ============================================================================

# Add to existing <IfModule mod_headers.c> section:

# Permissions-Policy (formerly Feature-Policy)

# Restricts which browser features can be used

Header always set Permissions-Policy "geolocation=(), microphone=(), camera=(), payment=()"

# X-Permitted-Cross-Domain-Policies

# Controls how PDFs from your domain can be embedded elsewhere

Header always set X-Permitted-Cross-Domain-Policies "none"

# Expect-CT (Certificate Transparency)

Header always set Expect-CT "max-age=86400, enforce"

Options -Indexes
</Directory>

# ============================================================================

# PREVENT HOTLINKING OF IMAGES

# ============================================================================

<IfModule mod_rewrite.c>
    RewriteCond %{HTTP_REFERER} !^$
    RewriteCond %{HTTP_REFERER} !^https://eslfunonline.com [NC]
    RewriteCond %{HTTP_REFERER} !^https://www.eslfunonline.com [NC]
    RewriteRule \.(jpg|jpeg|png|gif|webp)$ - [L,NC,F]
</IfModule>
